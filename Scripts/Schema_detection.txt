--CONVERT THE JSON DATA OF COLUMN menu_item_health_metrics_obj INTO STRUCTURED FORMAT

select 
m.* EXCLUDE (menu_item_health_metrics_obj),
j.value::string as ingredient
from
MENU_STAGING m,
LATERAL FLATTEN(input=>m.menu_item_health_metrics_obj:menu_item_health_metrics , recursive=>true) i,
LATERAL FLATTEN(input=>i.value:ingredients, recursive=>true) j,
WHERE m.MENU_ID=10001;
/*
10001	1	Ice Cream	Freezing Point	10	Lemonade	Beverage	Cold Option	0.6500	3.5000	Lemons
10001	1	Ice Cream	Freezing Point	10	Lemonade	Beverage	Cold Option	0.6500	3.5000	Sugar
10001	1	Ice Cream	Freezing Point	10	Lemonade	Beverage	Cold Option	0.6500	3.5000	Water
*/

select 
m.* EXCLUDE (menu_item_health_metrics_obj),
f.value:ingredients:: array as ingredients,
f.value:is_dairy_free_flag::string as is_dairy_free_flag,
f.value:is_gluten_free_flag::string as is_gluten_free_flag,
f.value:is_healthy_flag::string as is_healthy_flag,
f.value:is_nut_free_flag::string as is_nut_free_flag
from
MENU_STAGING m,
LATERAL FLATTEN(input=>m.menu_item_health_metrics_obj:menu_item_health_metrics 
-- ,MODE => 'ARRAY'
-- ,RECURSIVE => TRUE
) f
WHERE m.MENU_ID=10001 
/*

*/
